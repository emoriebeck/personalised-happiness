4.1 Model Function

mlVAR_call <- function(d, study, type){
  mlVAR_fun(d, study, type, 1)
  # mlVAR_fun(d, study, type, 2)
  return(NULL)
}

mlVAR_fun <- function(d, study, type, lag){
  vs <- colnames(d %>% select(-SID, -year))
  m <- mlVAR(
    d
    , vars = vs
    , idvar = "SID"
    , lags = lag
    # , beepvar = "year"
    , temporal = "correlated"
    , contemporaneous = "orthogonal"
    , verbose = TRUE
    , scale = T
  )
  save(m, file = sprintf("%s/04_results/01_mlVAR/%s-%s-lag%s.RData"
                         , wd, study, type, lag))
  return(NULL)
}

prep_fun <- function(d){
  cutoff <- 5
  na_test <- apply(d, 2, function(x) sum(!is.na(x)))
  if(any(na_test < cutoff)) return(NA) else return(d)
}

4.4 Plots

plot_fun <- function(x, type, sd = F, sig = "hide", study, h, lag){
  groups <- list(
    lifesat = which(x$fit$var == "lifesat")
    , other = seq(1:nrow(x$fit))[-which(x$fit$var == "lifesat")]
    )
  # ttl <- sprintf("\n%s: %s %s", study, h, lag)
  ttl <- study
  plot.res <- plot(x
                   , type
                   , layout = "spring"
                   , groups = groups
                   , nonsig = sig
                   , curve = -1
                   , legend = FALSE
                   , SD = sd
                   , details = FALSE
                   , mar = rep(10, 4)
                   , border.width = 2.5
                   # , title =  ttl
                   # , threshold = .01
                   , loop = .7
                   , node.width = 1.6
                   , edge.width = 1
                   , asize = 5
                   , label.font = 2
                   , label.fill.vertical = 1
                   , label.fill.horizontal = 1
                   , edge.color = "blue"
                   , DoNotPlot = TRUE
                   , color = c(RColorBrewer::brewer.pal(3, "Set3")[1], "white")
                   )
  if(sd == F){
  plot.res$graphAttributes$Edges$lty[plot.res$Edgelist$weight < 0] <- 2
  plot.res$graphAttributes$Edges$color <-
    ifelse(plot.res$Edgelist$weight <.02, "thistle2",
    ifelse(plot.res$Edgelist$weight <.04, "mediumorchid", "midnightblue"))
  }
  # plot(plot.res); title(ttl)
  return(list(p = plot.res, ttl = ttl))
}

net_nested <- net_nested %>%
