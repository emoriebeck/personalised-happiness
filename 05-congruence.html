edge_fun <- function(id, study){
  id_mat <- id %>% 
    select(SID, var1, op, var2, est) %>%
    unite(path, var1, op, var2, sep = "") %>%
    pivot_wider(names_from = "SID", values_from = "est")
  pop_mat <- net_nested %>%
    filter(Study == study) %>%
    select(lagged) %>%
    unnest(lagged) %>% 
    mutate(fixed = ifelse(P < .05, fixed, 0)) %>% 
    unite(path, to, from, sep = "<-") %>%
    select(path, pop = fixed) 
  full_mat <- id_mat %>% full_join(pop_mat)
  res <- sapply(full_mat %>% select(-path, -pop), function(x) {
    nEdges <- length(x)
    nEdgesOrig <- sum(full_mat$pop != 0, na.rm = T)
    nShared <- sum(full_mat$pop != 0 & x != 0, na.rm = T)
    nMissing <- sum(full_mat$pop == 0 & x == 0, na.rm = T) # missing edges
    tibble(prop_shared = nShared/nEdgesOrig,
         prop_missing = nMissing/(nEdges  - nEdgesOrig),
         prop_repl = prop_shared + prop_missing)
  }) %>%
    t() %>%
    data.frame() %>%
    rownames_to_column("SID") %>%
    as_tibble() %>%
    unnest(cols = c(prop_shared, prop_missing, prop_repl))
  return(res)
}

distance_fun <- function(id, study){
  # change format from long to wide (paths as columns)
  # merge together idiographic and population-level results
  id_mat <- id %>% 
    select(SID, var1, op, var2, est) %>%
    unite(path, var1, op, var2, sep = "") %>%
    full_join(
      net_nested %>%
      filter(Study == study) %>%
      select(lagged) %>%
      unnest(lagged) %>% 
      mutate(fixed = ifelse(P < .05, fixed, 0)) %>% 
      unite(path, to, from, sep = "<-") %>%
      select(path, est = fixed) %>%
      mutate(SID = "population")
      ) %>%
    pivot_wider(names_from = "path", values_from = "est")
  
  # estimate distance
  dist_mat <- dist(id_mat %>% select(-SID), upper = T) %>% as.matrix()
  colnames(dist_mat) <- id_mat$SID; rownames(dist_mat) <- id_mat$SID
  
  # get per-person average distance
  diag(dist_mat) <- NA
  mean_dist <- apply(dist_mat, 1, mean, na.rm = T)
  dist_mat[upper.tri(dist_mat)] <- NA
  
  # change distance matrix to long 
  long_dist <- dist_mat %>% 
    data.frame() %>%
    rownames_to_column("SID1") %>%
    pivot_longer(
      cols = -SID1
      , names_to = "SID2"
      , names_prefix = "X"
      , values_to = "edist"
      , values_drop_na = T
    ) %>%
    filter(SID1 != SID2)
  return(list(mean_dist, long_dist))
}
