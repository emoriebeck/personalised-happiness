plot_ts_fun <- function(d, study, hrmn, sid){
  p <- d %>%
    mutate(t = 1:n()) %>%
    pivot_longer(
      cols = -t
      , values_to = "value"
      , names_to = "var"
    ) %>%
    mutate(var = factor(var)
           , var = forcats::fct_relevel(var, "lifesat")) %>%
    ggplot(aes(x = t, y = value, color = var)) + 
      geom_line() + 
      geom_point(color = "black", size = .75, shape = "square") + 
      facet_grid(var~.) + 
      theme_classic() + 
      theme(legend.position = "none")
  # ggsave(p, file = sprintf("%s/04_results/04_figures/timeseries/%s-%s-%s.png", wd, study, hrmn, sid))
  ggsave(p, file = sprintf("%s/04_results/04_figures/timeseries/%s/%s/%s.pdf", wd, hrmn, study, sid), width = 8, height = 6)
}

sVAR_fun <- function(d, SID, study, type, dir, perc){
  print(sprintf("%s Subject %s, %.2f complete", study, SID, perc))
  
  # setup missings
  n <- nrow(d)
  cutoff <- ifelse(study == "GSOEP", 15, 5)
  na_test <- apply(d, 2, function(x) sum(is.na(x) == F))
  sd_test <- apply(d, 2, function(x) sd(x, na.rm = T) == 0)
  na_test <- which(na_test < cutoff | sd_test == TRUE)
  if(length(na_test >= 1)){d <- select(d, -na_test)}
  
  # setup whitelist
  cols <- colnames(d); ncols <- ncol(d)
  mat <- matrix(rep(1, ncols^2), nrow = ncols)
  colnames(mat) <- cols; rownames(mat) <- cols
  if(dir == "bottomup"){
    mat[rownames(mat) != "lifesat", colnames(mat) == "lifesat"] <- 0
  } else if(dir == "topdown"){
    mat[rownames(mat) == "lifesat", colnames(mat) != "lifesat"] <- 0
  } else if(dir == "nondirectional"){
    mat[rownames(mat) == "lifesat", ] <- 0
    mat[, colnames(mat) == "lifesat"] <- 0
  }
  
  fit <- 
    psychonetrics::var1(
      data = d
      , contemporaneous = "ggm"
      , beta = mat
      , missing = "pairwise"
      , standardize = "z"
    ) %>%
    psychonetrics::runmodel() %>%
    psychonetrics::prune(alpha = .05) %>%
    psychonetrics::stepup(criterion = "bic")
  
  beta <- fit@modelmatrices$fullsample$beta
  colnames(beta) <- cols; rownames(beta) <- cols
  if(dir %in% c("topdown", "nondirectional")){
    inval <- which(beta[colnames(beta) == "lifesat", ] != 0)
    inval <- inval[names(inval) !="lifesat"]
    if(length(inval) > 0){
      for(i in inval){
        fit <- fit %>%
          psychonetrics::fixpar("beta", which(cols == "lifesat"), i)
      }
      fit <- fit %>% 
        psychonetrics::runmodel()
      }
  } 
  if(dir %in% c("bottomup", "nondirectional")){
    inval <- which(beta[, colnames(beta) == "lifesat"] != 0)
    inval <- inval[names(inval) !="lifesat"]
    if(length(inval) > 0){
      for(i in inval){
        fit <- fit %>%
          psychonetrics::fixpar("beta", which(cols == "lifesat"), i)
      }
      fit <- fit %>% 
        psychonetrics::runmodel()
      }
  } 
    
  fitind <- psychonetrics::fit(fit) %>%
    data.frame() #%>%
    # filter(Measure %in% c("aic.x", "bic"))
  
  params <- psychonetrics::parameters(fit)
            
  # if("error" %in% class(fit)) return(fit)
  save(fit, file = sprintf("%s/04_results/06_svar/models/%s/%s/%s-%s.RData"
                           , wd, type, study, SID, dir))
  save(fitind, file = sprintf("%s/04_results/06_svar/fitind/%s/%s/%s-%s.RData"
                           , wd, type, study, SID, dir))
  save(params, file = sprintf("%s/04_results/06_svar/params/%s/%s/%s-%s.RData"
                           , wd, type, study, SID, dir))
  rm(list = c("params", "fit", "fitind", "d"))
  gc()
  return(T)
}

idio_tab_fun <- function(study, sid, dir, params){
  dir2 <- mapvalues(dir, c("topdown", "bottomup", "bidirectional", "nondirectional"),
                    c("Top-down", "Bottom-up", "Bi-directional", "Non-directional"), warn_missing = F)
  cap <- sprintf("<strong>Supplementary Table</strong><br><em>%s: All paths from both lagged and contemporaneous graphical VAR models for Participant %s in %s</em>", dir2, sid, study)
  
  tab <- params %>% 
    filter(op %in% c("<-", "--")) %>%
    select(V1 = var1, op, V2 = var2, est) %>%
    mutate(type = ifelse(op == "<-", "Lagged", "Contemporaneous"),
           type = factor(type, c("Lagged", "Contemporaneous")))
  
  rs <- tab %>% group_by(type) %>% tally() %>% 
    mutate(end = cumsum(n), start = lag(end) + 1, start = ifelse(is.na(start), 1, start))
  
  tab <- tab %>%
    select(-type) %>%
    kable(.
          , "html"
          , align = c(rep("c",4))
          , digits = 3
          , col.names = c("V1", "op", "V2", "Est.")
          , caption = cap
          # , escape = F
          ) %>%
    kable_classic(full_width = F, html_font = "Times") %>%
    kableExtra::group_rows(rs$type[1], rs$start[1], rs$end[1]) %>%
    kableExtra::group_rows(rs$type[2], rs$start[2], rs$end[2]) %>%
    footnote('"op" = Operator; -> = lagged partial correlation between V1 and V2; ~~ = contemporaneous partial correlation between V1 and V2.')
  save_kable(tab, file = sprintf("%s/04_results/03_tables/model-tabs/Harmonized/%s/%s-%s.html", wd, study, sid, dir))
  return(T)
}

info_crit_fun <- function(d, study){
  set.seed(12)
  d %>%
    filter(SID %in% sample(d$SID, 200)) %>%
    kable(.
          , "html"
          , digits = 2
          , cap = sprintf("<strong>Supplemental Table</strong><br><em>Bayesian Information Criterion from Graphical VAR Models for Four Different Sets of Theoretically-Relevant Models for %s", study)
          , col.names = c("Participant ID", "Non-Directional", "Bottom-Up", "Top-Down", "Bi-Directional")
          , align = rep("c", 5)
          ) %>%
    kable_classic(full_width = F, html_font = "Times") %>%
    scroll_box() %>%
    footnote("Each column refers to a separate graphical VAR model for the same participant with different path constraints. Specifically, we tested graphical VAR models for each participant in which all paths could be freely estimated (bi-directional), bottom-up paths were forced to be zero (top-down), top-down effects were forced to be zero (bottom-up), and both bottom-up and top-down effects were forced to zero (non-directional).") %>%
    save_kable(file = sprintf("%s/04_results/03_tables/information-criteria/%s.html", wd, study))
  return(T)
}

edge_colors <- RColorBrewer::brewer.pal(8, "Purples")[c(3,5,7)]
plot_fun <- function(data, SID, type, dataset){
  vars <- unique((data %>% filter(op == "<-"))$var2)
  vars <- c("lifesat", vars[vars != "lifesat"])
  x <- if(type == "Temporal") data %>% filter(op == "<-") else data %>% filter(op == "--")
  x <- x %>% 
    select(from = var2, to = var1, est) %>%
    mutate(from = factor(from)
           , from = forcats::fct_relevel(from, "lifesat")
           ) %>%
    arrange(from)
  groups <- factor(vars, levels = vars, labels = c("lifesat", rep("other", length(vars)-1)))
  plot <- qgraph(x
                 , layout = "spring" 
                 , title = sprintf("%s: %s for S%s", dataset, type, SID)
                 , loop = .7
                 , node.width = 1.4
                 , edge.width = 1
                 , label.font = 2
                 , label.fill.vertical = 1
                 , label.fill.horizontal = 1
                 , edge.color = "black"
                 , directed = if(type == "Temporal") T else F
                 , groups = groups
                 , color = c(RColorBrewer::brewer.pal(3, "Set3")[1], "white")
                 , legend = F
                 , DoNotPlot = TRUE
                 , mar = c(4,4,4,4)
                 , asize = 6
                 , border.width = 2.5
                 )
  plot$graphAttributes$Edges$lty[plot$Edgelist$weight < 0] <- 2
  plot$graphAttributes$Edges$color <-
    ifelse(abs(plot$Edgelist$weight) <.1, edge_colors[1],
    ifelse(abs(plot$Edgelist$weight) <.2, edge_colors[2], edge_colors[3]))
  return(plot)
}
