fun_call <- function(x, rule){
    switch(rule,
           average = mean(x, na.rm = T),
           mode = Mode(x)[1],
           sum = sum(x, na.rm = T),
           skip = unique(x)[1],
           select = unique(x)[1],
           max = max(x, na.rm = T),
           min = min(x, na.rm = T))
  }

  keep <- c("SIDp", "pidp", sprintf("%shid", WL))

r_fun <- function(dat, sample){
  dat <- dat %>% ungroup()
  r <- cor(dat %>% select(-SID), use = "pairwise")
  r[upper.tri(r, diag = F)] <- NA
  diag(r) <- 999
  nvar <- nrow(r)
  tab <- data.frame(r) %>% mutate(v1 = paste(1:nvar, ". ", rownames(.), sep = "")) %>% 
    select(v1, everything()) %>%
    full_join(dat %>% select(-SID) %>% gather(key = v1, value = value) %>%
                group_by(v1) %>%
                summarize(mean = mean(value, na.rm = T), sd = sd(value, na.rm = T)) %>%
                ungroup() %>%
                mutate(v1 = factor(v1, levels = colnames(r))) %>% 
                arrange(v1) %>%
                mutate(v1 = paste(1:nvar, ". ", v1, sep = ""))) %>%
    setNames(c(" ", 1:(nvar), "M", "SD")) %>%
    mutate_at(vars(`1`:SD), ~ifelse(!is.na(.) == T, sprintf("%.2f", .), ""))
  
  tab$M[grepl("gender", tab$v1)] <- paste(sprintf("%.2f", table(dat$gender)[1]/sum(table(dat$gender))*100), " %", sep = "")
  tab$SD[grepl("gender", tab$v1)] <- ""
  tab[tab == "999.00"] <- "--"
  
  ktab <- tab %>%
    kable(.
          , "html"
          , escape = F
          , align = c("l", rep("c", ncol(tab)-1))
          , cap = sprintf("<strong>Table SX</strong><br><em>Zero-Order Correlations and Descriptives for Baseline Measures in %s", sample)
          ) %>%
    kable_classic(full_width = F, html_font = "Times New Roman")
  save_kable(ktab, file = sprintf("%s/04_results/03_tables/zero-order-cors/%s.html", wd, sample))
  return(ktab)
}

n_tab_fun <- function(d, sample){
  n_sum <- d %>%
    group_by(year) %>%
    summarize_at(vars(-SID), ~sum(!is.na(.))) %>%
    arrange(year)
  tab <- n_sum %>%
    kable(.
          , "html"
          , escape = F
          , col.names = str_to_title(colnames(n_sum))
          , align = rep("c", ncol(n_sum))
          , caption = sprintf("<strong>Table SX</strong><br><em>%s Samples Sizes for Each Indicator Across Waves</em>", sample)
          ) %>%
    kable_classic(full_width = F, html_font = "Times New Roman")
  save_kable(tab, file = sprintf("%s/04_results/03_tables/sample-sizes/%s.html", wd, sample))
  return(tab)
}
